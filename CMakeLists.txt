cmake_minimum_required(VERSION 3.22)
project(SessionPhotos VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 add_compile_options(/utf-8)
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG v0.18.3 GIT_SHALLOW TRUE)
FetchContent_Declare(qrcode GIT_REPOSITORY https://github.com/nayuki/QR-Code-generator.git GIT_TAG v1.8.0 GIT_SHALLOW TRUE)
set(HTTPLIB_COMPILE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
foreach(dep JUCE httplib qrcode)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${dep}")
  string(TOUPPER "${dep}" upper)
  set(FETCHCONTENT_SOURCE_DIR_${upper} "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${dep}")
 endif()
endforeach()
FetchContent_MakeAvailable(JUCE httplib qrcode)
juce_add_binary_data(PhonePage SOURCES Web/index.html)
add_library(PhotoCore INTERFACE)
target_sources(PhotoCore INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source/PhotoStore.cpp ${CMAKE_CURRENT_SOURCE_DIR}/Source/LocalServer.cpp ${qrcode_SOURCE_DIR}/cpp/qrcodegen.cpp)
target_include_directories(PhotoCore INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source ${qrcode_SOURCE_DIR}/cpp)
target_compile_definitions(PhotoCore INTERFACE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(PhotoCore INTERFACE juce::juce_gui_basics httplib::httplib PhonePage)
set(SESSIONPHOTOS_FORMATS VST3)
if(APPLE)
 list(APPEND SESSIONPHOTOS_FORMATS AU)
endif()
set(SESSIONPHOTOS_AAX_SDK "" CACHE PATH "Path to an authorized Avid AAX SDK (optional)")
if(SESSIONPHOTOS_AAX_SDK)
 juce_set_aax_sdk_path("${SESSIONPHOTOS_AAX_SDK}")
 list(APPEND SESSIONPHOTOS_FORMATS AAX)
endif()
juce_add_plugin(SessionPhotos COMPANY_NAME "Benjamin Mandeau" BUNDLE_ID "fr.sessionphotos.plugin" PLUGIN_MANUFACTURER_CODE BMan PLUGIN_CODE Sph1 FORMATS ${SESSIONPHOTOS_FORMATS} AAX_IDENTIFIER "fr.sessionphotos.aax" PRODUCT_NAME "Session Photos" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Tools VST3_AUTO_MANIFEST FALSE)
target_sources(SessionPhotos PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_link_libraries(SessionPhotos PRIVATE PhotoCore juce::juce_audio_processors PUBLIC juce::juce_recommended_config_flags)
target_compile_definitions(SessionPhotos PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0)
option(SESSIONPHOTOS_TESTS "Build state and network integration tests" ON)
if(SESSIONPHOTOS_TESTS)
 enable_testing()
 add_executable(PhotoTests Tests/PhotoTests.cpp)
 target_include_directories(PhotoTests PRIVATE Source ${JUCE_SOURCE_DIR}/modules)
 target_link_libraries(PhotoTests PRIVATE SessionPhotos httplib::httplib)
 add_test(NAME state_and_network COMMAND PhotoTests)
endif()
